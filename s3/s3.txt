1. Setup (same pattern as SES)
bash
npm i @aws-sdk/client-s3 @aws-sdk/s3-request-presigner dotenv

.env:

AWS_ACCESS_KEY_ID=your_key
AWS_SECRET_ACCESS_KEY=your_secret
AWS_REGION=ap-south-1
S3_BUCKET_NAME=your-bucket-name


2. Basic — Create a bucket + upload/download/delete a file
javascript
import "dotenv/config";
import {
  S3Client,
  CreateBucketCommand,
  PutObjectCommand,
  GetObjectCommand,
  DeleteObjectCommand,
  ListObjectsV2Command,
} from "@aws-sdk/client-s3";
import fs from "fs";

const s3 = new S3Client({
  region: process.env.AWS_REGION,
  credentials: {
    accessKeyId: process.env.AWS_ACCESS_KEY_ID,
    secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY,
  },
});

const BUCKET = process.env.S3_BUCKET_NAME;

// Upload a file
async function uploadFile(filePath, key) {
  const fileContent = fs.readFileSync(filePath);
  const command = new PutObjectCommand({
    Bucket: BUCKET,
    Key: key, // e.g. "images/profile.png"
    Body: fileContent,
    ContentType: "image/png", // adjust based on file type
  });
  const res = await s3.send(command);
  console.log("Uploaded:", key, res);
}

// List objects
async function listFiles() {
  const command = new ListObjectsV2Command({ Bucket: BUCKET });
  const res = await s3.send(command);
  console.log(res.Contents?.map((obj) => obj.Key));
}

// Delete a file
async function deleteFile(key) {
  const command = new DeleteObjectCommand({ Bucket: BUCKET, Key: key });
  await s3.send(command);
  console.log("Deleted:", key);
}

await uploadFile("./test.png", "images/test.png");
await listFiles();

3. Intermediate — Presigned URLs (very relevant for your backend work)

Instead of exposing your AWS keys or making the bucket public, generate a temporary signed URL the frontend can use directly:

javascript
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";
import { PutObjectCommand, GetObjectCommand } from "@aws-sdk/client-s3";

// Give frontend a temporary URL to upload directly to S3
async function getUploadUrl(key, contentType) {
  const command = new PutObjectCommand({
    Bucket: BUCKET,
    Key: key,
    ContentType: contentType,
  });
  const url = await getSignedUrl(s3, command, { expiresIn: 300 }); // 5 min
  return url;
}

// Give frontend a temporary URL to download/view a private file
async function getDownloadUrl(key) {
  const command = new GetObjectCommand({ Bucket: BUCKET, Key: key });
  const url = await getSignedUrl(s3, command, { expiresIn: 3600 }); // 1 hr
  return url;
}

This is the standard pattern for things like profile picture uploads: your Express backend generates a presigned URL, sends it to the frontend, and the browser uploads directly to S3 — your server never touches the raw file bytes. Scales much better than routing uploads through your server.
